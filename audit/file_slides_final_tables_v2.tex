% Backup: protocol caveats the audience may ask about.
\begin{frame}{Backup: Baseline Protocol Notes}
\small
\begin{itemize}
  \setlength{\itemsep}{3pt}
  \item All methods start from the same Panoptic-FPN R50 Cityscapes checkpoint;
        strict online, batch size 1, no resets.
  \item TENT, CoTTA, OBAO-seg and DAT-BF adapt segmentation only;
        AMROD and CTAOD adapt detection only.
        Their numbers are task context, not joint multi-task results.
  \item DAT-BF is a same-source Panoptic-FPN port, not an official DAT reproduction.
        OBAO-seg official-code parity audit is still open; no long-term OBAO runs.
  \item CT-CR + Task-CoCo uses the same CT-CR loss as CTCMT and adds gradient
        coordination: it compares optimisation strategies, not a CT-CR-free baseline.
  \item Single-seed baselines: TENT, CoTTA, OBAO-seg, DAT-BF, AMROD, CTAOD long-term.
        CTAOD CSC-LT is the rerun (first run: 13.49).
\end{itemize}
\end{frame}
